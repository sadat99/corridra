// Captures full-page screenshots and computed colours for AC/1 before/after review.
import { chromium } from '@playwright/test';
const [, , base, label] = process.argv;
const routes = ['/', '/how-it-works', '/corridors', '/trust', '/signin', '/plan', '/operator/signin', '/workspace', '/institutions'];
const widths = [390, 768, 1440];
const browser = await chromium.launch({ executablePath: '/opt/pw-browsers/chromium' });
const report = [];
for (const width of widths) {
  const page = await browser.newPage({ viewport: { width, height: 900 } });
  for (const route of routes) {
    const res = await page.goto(base + route, { waitUntil: 'networkidle' });
    const name = (route === '/' ? 'home' : route.slice(1).replaceAll('/', '_')) + `_${width}`;
    await page.screenshot({ path: `work/ac1-shots/${label}/${name}.png`, fullPage: true });
    const probe = await page.evaluate(() => {
      const pick = (sel) => {
        const el = document.querySelector(sel);
        if (!el) return null;
        const s = getComputedStyle(el);
        return { bg: s.backgroundColor, fg: s.color, border: s.borderColor };
      };
      return {
        overflow: document.documentElement.scrollWidth > window.innerWidth,
        body: pick('body'),
        primary: pick('.primary-link'),
        band: pick('.guide-band'),
        link: pick('.text-link'),
      };
    });
    report.push({ route, width, status: res.status(), ...probe });
  }
  await page.close();
}
await browser.close();
console.log(JSON.stringify(report));
